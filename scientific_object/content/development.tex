\section{Стандарты

\subsection{Стиль (Style
Guide)}\label{ux441ux442ux438ux43bux44c-style-guide}

\begin{itemize}
\tightlist
\item
  \textbf{Индентация}: 2 пробела (табы запрещены).
\item
  \textbf{Длина строки}: макс. 100 символов.
\item
  \textbf{Предупреждения}: Весь код должен компилироваться с
  \texttt{-Wall} без предупреждений.
\end{itemize}

\subsection{Именование}\label{ux438ux43cux435ux43dux43eux432ux430ux43dux438ux435}

\begin{itemize}
\tightlist
\item
  \textbf{Модули}: PascalCase (\texttt{Core.Tax})
\item
  \textbf{Типы/Конструкторы}: PascalCase (\texttt{TaxRate},
  \texttt{Asset})
\item
  \textbf{Функции/Поля}: camelCase (\texttt{calcVAT}, \texttt{trRate})
\end{itemize}

\subsection{Работа с
типами}\label{ux440ux430ux431ux43eux442ux430-ux441-ux442ux438ux43fux430ux43cux438}

\begin{itemize}
\tightlist
\item
  Используйте \texttt{Text} вместо \texttt{String}.
\item
  Используйте \texttt{Int64} для ID из базы данных.
\item
  Используйте \texttt{Newtype} для оберток доменных понятий
  (\texttt{Money}, \texttt{BillID}).
\end{itemize}

\section{Завершение сессии (Session
Completion)}\label{ux437ux430ux432ux435ux440ux448ux435ux43dux438ux435-ux441ux435ux441ux441ux438ux438-session-completion}

\textbf{ВАЖНО}: Работа считается завершенной только после успешного
выполнения \texttt{git\ push}.

\subsection{Обязательный
процесс:}\label{ux43eux431ux44fux437ux430ux442ux435ux43bux44cux43dux44bux439-ux43fux440ux43eux446ux435ux441ux441}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Зафиксировать остаток работ}: Создать задачи в Beads для
  всего, что требует продолжения.
\item
  \textbf{Проверка качества}: Запустить тесты и линтеры.
\item
  \textbf{Обновить статус задач}: Закрыть выполненные задачи в Beads.
\item
  \textbf{PUSH В РЕПОЗИТОРИЙ}:
  \texttt{bash\ \ \ \ \ git\ pull\ -\/-rebase\ \ \ \ \ git\ push}
\item
  \textbf{Верификация}: Убедиться, что \texttt{git\ status} показывает
  ``up to date with origin''.
\end{enumerate}
